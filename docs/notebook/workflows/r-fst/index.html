\(F_{ST}\) and \(F_{IS}\) with simulated forest populations

Simulating several forests stands of trees over 150 simualtions at once SNP. Each tree’s genotype in the simulation is represented by one number.

Code Genotype Number of A alleles Description
0 aa 0 Homozygous a
1 Aa 1 Heterozygous Aa
2 AA 2 Homozygous A

For example:

g <- c(0,1,2,1)

represents four trees:

aa Aa AA Aa

Scenarios and Expected results

Scenario What changes What to expect
A. Drift
50 trees/stand		 Isolated stands that randomly mates Allele frequencies drift apart. \(F_{ST}\) rises towards 1 and \(F_{IS}\) stays near 0.
B. Gene flow 5% of the pollen comes from other stands \(F_{ST}\) stays low.
C. Selfing Half of the seeds comes from self-pollination. \(F_{IS}\) stays above 0, close to the theory value \(s / (2 - s)\), which is 0.33 when \(s = 0.5\).
D. Drift
500 trees/stand		 Larger stands \(F_{ST}\) rises much more slowly thn in Scenario A.

2. Calculate allele frequency

allele_freq <- function(g) {sum(g) / (2 * length(g))}

This defines a function named allele_freq(). It’s input g is a vector of genotypes. The numerator of this function adds the number of A alleles across all trees. While, in the denominator counts all the allele copies.

For examples: If there are four diploid trees, then they carry eight copies in total.

Therefore if g = c(0, 1, 2, 1) then;

\[sum(g) = 0 + 1 + 2 + 1 = 4\]
\[2 * length(g) = 2 \times 4 = 8\] \[p_{A} = \frac{4}{8} = 0.5\]

3. Calculate observed heterozygosity

obs_het <- function(g) { mean(g == 1)}

This functions checks which tree in the sample is heterozygous (g==1), which would return:

g==1
[1] FALSE  TRUE FALSE  TRUE

When calclating a mean, R treats TRUE as 1 and FALSE as 0: \[H_{O} = \frac{0 + 1 + 0 + 1}{4} = 0.5\] Thus the observed heterozygosity (\(H_{O}\)) is simply the fraction of heterozygous trees.

4. Calculate expected heterozygosity

exp_het <- function(p) { 2 * p * (1 - p) }

Given the allele frequency \(p\), this calculates:

\[H_{E} = 2p(1 - p)\] For \(p = 0.5\) :

\[ H_{E} = 2(0.5)(1 - 0.5) = 0.5\]

The important distinction between the two heterozygosity measures is:

  • Observed heterozygosity (\(H_{O}\)) comes from the actual genotype counts.

  • Expected heterozygosity (\(H_{E}\)) comes from the allele frequency and the model’s random mating reference.

5. Define F-statistic calculation

# F-statistics are undefined when their denominator is zero.
safe_F <- function(observed, expected) {
                        if (expected > 0) {1 - observed / expected} 
                                                
                        else {NA_real_}
                                        }
# -------------------------------------------------------------
# 2. Reproduction within one stand
# -------------------------------------------------------------

make_offspring <- function(
    parents,
    N,
    selfing = 0,
    m = 0,
    migrant_p = NA_real_
) {

  # Draw one allele from each selected parent.
  gamete <- function(idx) {
    rbinom(
      n = length(idx),
      size = 1,
      prob = parents[idx] / 2
    )
  }

  # Maternal contribution.
  mom <- sample(
    seq_along(parents),
    size = N,
    replace = TRUE
  )

  egg <- gamete(mom)

  # Designated selfing versus random local mating.
  is_self <- runif(N) < selfing

  dad <- ifelse(
    is_self,
    mom,
    sample(seq_along(parents), N, replace = TRUE)
  )

  pollen <- gamete(dad)

  # Migrant pollen replaces some non-designated-self pollen.
  if (m > 0) {

    from_out <- (runif(N) < m) & !is_self

    pollen[from_out] <- rbinom(
      n = sum(from_out),
      size = 1,
      prob = migrant_p
    )
  }

  egg + pollen
}

# -------------------------------------------------------------
# 3. Simulate multiple stands
# -------------------------------------------------------------

simulate_stands <- function(
    n_pop = 6,
    N = 50,
    n_gen = 100,
    p0 = 0.5,
    selfing = 0,
    m = 0
) {

  # Independent draws from the same starting distribution.
  # Realized starting allele frequencies vary among stands.
  stands <- replicate(
    n_pop,
    rbinom(N, size = 2, prob = p0),
    simplify = FALSE
  )

  freq <- matrix(
    NA_real_,
    nrow = n_gen + 1,
    ncol = n_pop,
    dimnames = list(NULL, paste0("Stand ", seq_len(n_pop)))
  )

  stats <- data.frame(
    gen = 0:n_gen,
    HI = NA_real_,
    HS = NA_real_,
    HT = NA_real_,
    FIS = NA_real_,
    FST = NA_real_,
    FIT = NA_real_
  )

  for (t in 0:n_gen) {

    p <- vapply(stands, allele_freq, numeric(1))

    HI <- mean(vapply(stands, obs_het, numeric(1)))
    HS <- mean(exp_het(p))
    HT <- exp_het(mean(p))

    freq[t + 1, ] <- p

    stats[t + 1, 2:7] <- c(
      HI,
      HS,
      HT,
      safe_F(HI, HS),
      safe_F(HS, HT),
      safe_F(HI, HT)
    )

    if (t == n_gen) break

    # Shared pollen pool includes all stands.
    pbar <- mean(p)

    stands <- lapply(
      stands,
      make_offspring,
      N = N,
      selfing = selfing,
      m = m,
      migrant_p = pbar
    )
  }

  list(
    freq = freq,
    stats = stats,
    final = stands
  )
}

freq_data <- do.call(
  rbind,
  lapply(names(sims), function(id) {

    sim <- sims[[id]]

    data.frame(
      scenario = id,
      generation = rep(
        sim$stats$gen,
        times = ncol(sim$freq)
      ),
      stand = rep(
        colnames(sim$freq),
        each = nrow(sim$freq)
      ),
      value = as.vector(sim$freq)
    )
  })
)

F_data <- do.call(
  rbind,
  lapply(names(sims), function(id) {

    s <- sims[[id]]$stats

    rbind(
      data.frame(
        scenario = id,
        generation = s$gen,
        statistic = "FST: among stands",
        value = s$FST
      ),
      data.frame(
        scenario = id,
        generation = s$gen,
        statistic = "FIS: within stands",
        value = s$FIS
      )
    )
  })
)

      panel.grid.major.y = element_line(

        linewidth = 0.3

      strip.background = element_rect(
        fill = "grey95",
        color = "grey75"
      ),
      strip.text = element_text(
        face = "bold",
        size = 10,
        margin = margin(7, 5, 7, 5)

      legend.box = "vertical",

      plot.caption = element_text(
        hjust = 0,

# 8. Compare replicate FST trajectories with a drift benchmark
# -------------------------------------------------------------

n_reps <- 20
N_theory <- 50

reps <- replicate(
  n_reps,
  simulate_stands(
    n_pop = 10,
    N = N_theory,
    n_gen = n_gen
  )$stats$FST
)

rep_data <- data.frame(
  generation = rep(0:n_gen, times = n_reps),
  replicate = factor(
    rep(seq_len(n_reps), each = n_gen + 1)
  ),
  FST = as.vector(reps)
)

# Avoid returning NaN when all replicates are undefined.
mean_FST <- apply(
  reps,
  1,
  function(x) {
    if (all(is.na(x))) NA_real_ else mean(x, na.rm = TRUE)
  }
)

curve_data <- rbind(
  data.frame(
    generation = 0:n_gen,
    FST = mean_FST,
    curve = "Mean of simulated runs"
  ),
  data.frame(
    generation = 0:n_gen,
    FST = 1 - (1 - 1 / (2 * N_theory))^(0:n_gen),
    curve = "Idealized drift benchmark"
  )
)
